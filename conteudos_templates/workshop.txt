// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC721/ERC721.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract WorkshopCertificate is ERC721, Ownable {
    uint256 public tokenIdCounter;
    mapping(address => bool) public hasMinted;

    constructor() ERC721("Workshop Solidity Certificate", "WSC") {}

    function mintCertificate() external {
        require(!hasMinted[msg.sender], "Certificate already minted");

        uint256 tokenId = tokenIdCounter;
        tokenIdCounter++;

        hasMinted[msg.sender] = true;
        _safeMint(msg.sender, tokenId);
    }
}

Explique apenas o essencial:

Por que usar mapping

Por que limitar 1 NFT por wallet

Por que Ownable (gancho pra evoluir depois)
